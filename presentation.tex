\begin{frame}{Spherical Voronoi: Volumes \& Cell Graphs}

            \begin{itemize}
                \item \textbf{Goal:} Assign realistic 3D cell volumes and build a cell interaction graph

                \begin{itemize}
                    \item Scatter cell centers randomly on the sphere surface

                    \item \textbf{Graph:} Connect an edge where Voronoi regions touch
                \end{itemize}
                \item Automatically encodes realistic spatial constraints:
                \begin{itemize}
                    \item Volume is driven by packing geometry
                    \item Graph edges naturally reflect cell–cell contacts
                \end{itemize}

% Slide: Ripley's Functions Explained
\begin{frame}{What are Ripley's Functions?}
    \textbf{Ripley's K, F, and G Functions}\\
    \vspace{1em}
    \begin{itemize}
        \item \textbf{Purpose:} Quantify spatial patterns of point data---are points \textit{clustered}, \textit{random}, or \textit{regularly spaced}?
        \item \textbf{How:} Measure how many points are within a distance $r$ of each other, compared to randomness.
    \end{itemize}
    \vspace{1em}
    \textbf{Key Functions:}
    \begin{itemize}
        \item $\mathbf{K}(r)$: Average number of other points within distance $r$ of a typical point.
        \item $\mathbf{F}(r)$: Probability the nearest point from a random location is within $r$.
        \item $\mathbf{G}(r)$: Probability the nearest neighbor of a point is within $r$.
    \end{itemize}
    \vspace{1em}
    \textbf{Interpretation:}\\
    Compare to random (Poisson) pattern:
    \begin{itemize}
        \item Above random $\rightarrow$ clustering
        \item Below random $\rightarrow$ regularity/repulsion
    \end{itemize}
\end{frame}

% Slide 16: Why Not 3D CNNs
\begin{frame}{Why Not 3D CNNs?}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Memory Constraints:}
            \begin{itemize}
                \item Single organoid: $\sim$2 GB
                \item $2048 \times 2048 \times 200$ voxels
                \item GPU memory: prohibitive
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Downsampling Issues:}
            \begin{itemize}
                \item Destroys cellular details
                \item Individual cells unresolvable
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{No Native Invariance:}
            \begin{itemize}
                \item Not invariant to 3D rotations
                \item Requires extensive augmentation
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Grid-Based Limitation:}
            \begin{itemize}
                \item Treats organoid as voxel grid
                \item Doesn't model cells as entities
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5em}
    \begin{alertblock}{Conclusion}
        Graphs + GNNs overcome all these limitations
    \end{alertblock}

% Slide 15: The Leap—From CNNs to GNNs
\begin{frame}{From CNNs to GNNs: A New Paradigm}
    \begin{columns}
        \begin{column}{0.54\textwidth}
            \textbf{CNNs}: Grids of pixels/voxels \\
            \hspace{0.5em} $\Rightarrow$ Convolutions aggregate over local neighborhoods\\
            \vspace{0.8em}
            \textbf{BUT:}\\
            \begin{itemize}
                \item Real organoids = collections of \textbf{cells}, not voxels
                \item \textbf{Spatial relations} matter (who's next to whom)
                \item Grids ignore true biological topology
            \end{itemize}
            \vspace{1em}
            \textbf{GNNs}: Process data as \textbf{graphs}
            \begin{itemize}
                \item \textbf{Nodes}: Cells\\
                \item \textbf{Edges}: Physical contact or spatial proximity\\
                \item \textbf{Features}: Position, volume, etc.
            \end{itemize}
            \vspace{1em}
            \textbf{Key Takeaway:} \\
            \textcolor{inriaBlue}{Graphs model the \textit{relational structure} of real organoids.}
        \end{column}

% Slide 17: Message Passing
\begin{frame}{GNNs: Message Passing Paradigm}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Core Idea:} Nodes exchange information
            
            \vspace{0.5em}
            \textbf{Update Rule:}
            \begin{equation*}
                h_i^{(l+1)} = \text{UPDATE}\left(h_i^{(l)}, \text{AGG}\left(\{h_j^{(l)}\}_{j \in \mathcal{N}(i)}\right)\right)
            \end{equation*}
            
            \vspace{0.5em}
            \textbf{Iterative process:}
            \begin{enumerate}
                \item Each node sends messages to neighbors
                \item Aggregate incoming messages
                \item Update node representation
            \end{enumerate}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.75]
                    \node[circle, draw, fill=inriaBlue!40, minimum size=1cm] (A) at (0,0) {$i$};
                    \node[circle, draw, fill=gray!30, minimum size=0.8cm] (B) at (-1.5,1.2) {$j_1$};
                    \node[circle, draw, fill=gray!30, minimum size=0.8cm] (C) at (1.5,1.2) {$j_2$};

\begin{frame}{From Node to Graph: Global Pooling}

% Slide 18: GCN
\begin{frame}{GCN: Graph Convolutional Network (Baseline)}
    \textbf{Kipf \& Welling, 2017}
    
    \vspace{0.5em}
    \textbf{Update Rule:}
    \begin{equation*}
        h_i^{(l+1)} = \sigma\left(\sum_{j \in \mathcal{N}(i) \cup \{i\}} \frac{1}{\sqrt{d_i d_j}} W^{(l)} h_j^{(l)}\right)
    \end{equation*}
    
    \vspace{0.5em}
    \textbf{Properties:}
    \begin{itemize}
        \item Normalized mean aggregation
        \item Simple and effective baseline
        \item \textcolor{red}{\faTimes\ Treats all neighbors equally}
        \item \textcolor{red}{\faTimes\ No distance/importance weighting}
    \end{itemize}
\end{frame}

% Slide 19: GAT
\begin{frame}{GAT: Graph Attention Network}
    \textbf{Veličković et al., 2018}
    
    \vspace{0.3em}
    \textbf{Key Idea:} Learn attention coefficients to weight neighbors
    
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{1. Compute attention scores:}
    \begin{equation*}
                \alpha_{ij} = \text{softmax}_j\left(\text{score}(h_i, h_j)\right)
    \end{equation*}
    
            \textbf{2. Weighted aggregation:}
    \begin{equation*}
                h_i^{\text{new}} = \sigma\left(\sum_{j \in \mathcal{N}(i)} \textcolor{inriaRed}{\alpha_{ij}} \cdot W h_j\right)
    \end{equation*}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.7]
                    \node[circle, draw, fill=inriaBlue!40, minimum size=0.8cm] (A) at (0,0) {$i$};

                \end{tikzpicture}\\
                {\footnotesize Learned attention weights}
            \end{center}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{DeepSets: Set-Based Baseline}
    \textbf{Zaheer et al., 2017}
    
    \vspace{0.5em}
    \textbf{Approach:} Treat organoid as unordered set of cells (ignore edges)

% Slide 22: EGNN
\begin{frame}{EGNN: Equivariant Graph Neural Network}
    \textbf{Satorras et al., 2021}
    
    \vspace{0.5em}
    \textbf{Key Principle:} Use only \textbf{invariant quantities} in messages
    
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Message} (uses distance = invariant):
    \begin{equation*}
        m_{ij} = \phi_m\left(h_i, h_j, \|x_i - x_j\|^2\right)
    \end{equation*}
    
    \textbf{Coordinate update} (equivariant):
    \begin{equation*}
                x_i^{new} = x_i + \sum_{j} (x_i - x_j) \cdot \phi_x(m_{ij})
    \end{equation*}
        \end{column}
        \begin{column}{0.45\textwidth}
            \textbf{Advantages:}
            \begin{itemize}
                \item \textcolor{cysticGreen}{\faCheck} No augmentation needed
                \item \textcolor{cysticGreen}{\faCheck} Guaranteed robustness
                \item \textcolor{cysticGreen}{\faCheck} Better with less data
            \end{itemize}
    
            \vspace{0.3em}
            \textbf{Trade-off:}\\
            Slightly lower raw accuracy
        \end{column}
    \end{columns}
\end{frame}
