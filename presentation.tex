% Slide 10: Limitations of 3D CNNs
\begin{frame}{Why Not 3D CNNs?}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Memory Constraints:}
            \begin{itemize}
                \item Single organoid: $\sim$2 GB
                \item $2048 \times 2048 \times 200$ voxels
                \item GPU memory: prohibitive
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Downsampling Issues:}
            \begin{itemize}
                \item Destroys cellular details
                \item Individual cells unresolvable
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{No Native Invariance:}
            \begin{itemize}
                \item Not invariant to 3D rotations
                \item Requires extensive augmentation
                \item No guarantees
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Grid-Based Limitation:}
            \begin{itemize}
                \item Treats organoid as voxel grid
                \item Doesn't model cells as entities
                \item No relational structure
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5em}
    \begin{alertblock}{Conclusion}
        3D CNNs are not adapted to our constraints
    \end{alertblock}
\end{frame}

% Slide 12: Introduction to GNNs
\begin{frame}{Graph Neural Networks: Message Passing}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Challenge:} Graphs are non-Euclidean
            \begin{itemize}
                \item Variable neighbors, no ordering
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Solution: Message Passing}
            \begin{equation*}
                h_i^{(l+1)} = \text{UPD}\left(h_i^{(l)}, \text{AGG}\left(\{h_j^{(l)}\}_{j \in \mathcal{N}(i)}\right)\right)
            \end{equation*}
            
            \vspace{0.5em}
            \textbf{Iterative process:}
            \begin{enumerate}
                \item Each node sends message to neighbors
                \item Aggregate incoming messages
                \item Update node representation
            \end{enumerate}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.75, node distance=1.5cm]
                    \node[circle, draw, fill=inriaBlue!40, minimum size=1cm] (A) at (0,0) {$i$};
                    \node[circle, draw, fill=gray!30, minimum size=0.8cm] (B) at (-1.5,1.2) {$j_1$};
                    \node[circle, draw, fill=gray!30, minimum size=0.8cm] (C) at (1.5,1.2) {$j_2$};
                    \node[circle, draw, fill=gray!30, minimum size=0.8cm] (D) at (0,-1.5) {$j_3$};
                    
                    \draw[->, thick, inriaRed, decorate, decoration={snake, amplitude=1pt}] (B) -- (A);
                    \draw[->, thick, inriaRed, decorate, decoration={snake, amplitude=1pt}] (C) -- (A);
                    \draw[->, thick, inriaRed, decorate, decoration={snake, amplitude=1pt}] (D) -- (A);
                \end{tikzpicture}\\
                {\footnotesize Messages flow to central node}\\[0.3em]
                {\tiny \faPlayCircle\ Animation: \texttt{message\_passing.gif}}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% Slide 13: Global Pooling
\begin{frame}{Global Pooling and Classification}
    \textbf{Goal:} Graph-level prediction (not node-level)
    
    \vspace{1em}
    \textbf{Global Pooling:}
    \begin{itemize}
        \item \textbf{Mean pooling:} $h_G = \frac{1}{|V|}\sum_{i \in V} h_i$
        \item \textbf{Max pooling:} $h_G = \max_{i \in V} h_i$
        \item \textbf{Attention pooling:} learned importance weights
    \end{itemize}
    
    \vspace{1em}
    \textbf{Classification Head:}
    \begin{equation*}
        \text{Node features} \xrightarrow{\text{GNN layers}} \text{Node embeddings} \xrightarrow{\text{Pooling}} h_G \xrightarrow{\text{MLP}} \hat{y}
    \end{equation*}
\end{frame}

% Slide 14: GCN Architecture
\begin{frame}{GCN: Graph Convolutional Network (Baseline)}
    \textbf{Kipf \& Welling, 2017}
    
    \vspace{0.5em}
    \textbf{Update Rule:}
    \begin{equation*}
        h_i^{(l+1)} = \sigma\left(\sum_{j \in \mathcal{N}(i) \cup \{i\}} \frac{1}{\sqrt{d_i d_j}} W^{(l)} h_j^{(l)}\right)
    \end{equation*}
    
    \vspace{0.5em}
    \textbf{Properties:}
    \begin{itemize}
        \item Normalized mean aggregation
        \item Simple and effective
        \item \textcolor{red}{Treats all neighbors equally}
        \item \textcolor{red}{No distance/importance weighting}
    \end{itemize}
\end{frame}

% Slide 15: GAT Architecture
\begin{frame}{GAT: Graph Attention Network (Main Choice)}
    \textbf{Veličković et al., 2018}
    
    \vspace{0.5em}
    \textbf{Key Idea:} Learn attention coefficients to weight neighbors differently
    
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{1. Compute attention scores:}
            \begin{equation*}
                \alpha_{ij} = \text{softmax}_j\left(\text{score}(h_i, h_j)\right)
            \end{equation*}
            
            \textbf{2. Weighted aggregation:}
            \begin{equation*}
                h_i^{\text{new}} = \sigma\left(\sum_{j \in \mathcal{N}(i)} \textcolor{inriaRed}{\alpha_{ij}} \cdot W h_j\right)
            \end{equation*}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.7]
                    \node[circle, draw, fill=inriaBlue!40, minimum size=0.8cm] (A) at (0,0) {$i$};
                    \node[circle, draw, fill=gray!30, minimum size=0.6cm] (B) at (-1.5,1.2) {$j_1$};
                    \node[circle, draw, fill=gray!30, minimum size=0.6cm] (C) at (1.5,1.2) {$j_2$};
                    \node[circle, draw, fill=gray!30, minimum size=0.6cm] (D) at (0,-1.5) {$j_3$};
                    
                    \draw[->, line width=2pt, inriaRed] (B) -- (A) node[midway, left] {\tiny 0.5};
                    \draw[->, line width=1pt, inriaRed!50] (C) -- (A) node[midway, right] {\tiny 0.2};
                    \draw[->, line width=1.5pt, inriaRed!75] (D) -- (A) node[midway, right] {\tiny 0.3};
                \end{tikzpicture}\\
                {\footnotesize Learned attention weights}
            \end{center}
        \end{column}
    \end{columns}
    
    \vspace{0.5em}
    \textcolor{cysticGreen}{\faCheck} \textbf{Best performance} — Multi-head attention for robustness
\end{frame}

% Slide 16: DeepSets
\begin{frame}{DeepSets: Set-Based Baseline}
    \textbf{Zaheer et al., 2017}
    
    \vspace{0.5em}
    \textbf{Approach:} Treat organoid as unordered set of cells
    
    \begin{equation*}
        f(X) = \rho\left(\sum_{x \in X} \phi(x)\right)
    \end{equation*}
    
    \vspace{0.5em}
    \textbf{Properties:}
    \begin{itemize}
        \item Permutation invariant
        \item \textcolor{red}{Ignores local spatial structure}
        \item Each cell processed independently before aggregation
    \end{itemize}
    
    \vspace{0.5em}
    \textbf{Observation:} Performs reasonably well\\
    $\Rightarrow$ Global statistics are informative, but local structure helps
\end{frame}

% Slide 19: EGNN Architecture
\begin{frame}{EGNN: Equivariant Graph Neural Network}
    \textbf{Satorras et al., 2021}
    
    \vspace{0.5em}
    \textbf{Key Principle:} Use only invariant quantities in messages
    \begin{itemize}
        \item Distances are invariant (don't change under rotation)
        \item Direction vectors are equivariant
    \end{itemize}
    
    \vspace{0.5em}
    \textbf{Practical Advantages:}
    \begin{itemize}
        \item \textcolor{cysticGreen}{\faCheck} No data augmentation needed
        \item \textcolor{cysticGreen}{\faCheck} Better generalization with less data
        \item \textcolor{cysticGreen}{\faCheck} Robustness guaranteed by construction
    \end{itemize}
    
    \vspace{0.5em}
    \textbf{Trade-off:} Slightly lower raw performance, but guaranteed geometric robustness
\end{frame}

% Slide 20: EGNN Formulas
\begin{frame}{EGNN: Key Formulas}
    \textbf{Message computation} (invariant):
    \begin{equation*}
        m_{ij} = \phi_m\left(h_i, h_j, \|x_i - x_j\|^2\right)
    \end{equation*}
    
    \textbf{Coordinate update} (equivariant):
    \begin{equation*}
        x_i^{(l+1)} = x_i^{(l)} + \sum_{j \neq i} (x_i - x_j) \cdot \phi_x(m_{ij})
    \end{equation*}
    
    \textbf{Feature update}:
    \begin{equation*}
        h_i^{(l+1)} = \phi_h\left(h_i^{(l)}, \sum_{j \in \mathcal{N}(i)} m_{ij}\right)
    \end{equation*}
    
    \vspace{0.5em}
    \begin{block}{Best of Both Worlds}
        Invariant messages + Equivariant coordinate updates
    \end{block}
\end{frame}

% Slide 34: GRETSI Study Protocol

    \textbf{Research Question:}\\
    When should we prefer GNNs over classical methods (Ripley's K)?
    
    \vspace{0.5em}
    \textbf{Protocol:}
    \begin{itemize}
        \item Synthetic data with controlled ground truth
        \item Spherical point distributions: Poisson vs Matérn
        \item Two noise types: Gaussian (position) + Salt-and-pepper (points)
    \end{itemize}
    
    \vspace{0.5em}
    \textbf{Comparison:}
    \begin{itemize}
        \item GNN classifiers (various depths)
        \item Random Forest on Ripley's K, F, G functions
    \end{itemize}

% Slide 35: Noise Robustness
\begin{frame}{Result 1: Noise Robustness}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{img/noise.png}
            \end{center}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Observation:}\\
            Spatial statistics are \textbf{more robust to noise}
            
            \vspace{0.5em}
            \textbf{Why?}
            \begin{itemize}
                \item Ripley's functions average over many point pairs
                \item Natural smoothing of noise
                \item GNNs propagate noise through layers
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Optimal GNN depth:} 5-6 layers
            \begin{itemize}
                \item Deeper $\rightarrow$ over-smoothing
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide 36: Geometric Generalization
\begin{frame}{Result 2: Geometric Generalization}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{img/accuracy_vs_ratio.png}
            \end{center}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Training:} Spherical distributions\\
            \textbf{Testing:} Ellipsoids (ratio 1:1 → 5:1)
            
            \vspace{0.5em}
            \textbf{Results:}
            \begin{itemize}
                \item Spatial stats: 100\% → 65\% (\textcolor{red}{$-$35\%})
                \item GNNs: 95\% → 82\% (\textcolor{cysticGreen}{$-$13\%})
            \end{itemize}
            
            \vspace{0.5em}
            \textbf{Why?}
            \begin{itemize}
                \item Ripley's K assumes isotropy
                \item GNNs learn topological patterns
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5em}
    \begin{alertblock}{Conclusion}
        GNNs excel when morphologies vary — exactly our situation!
    \end{alertblock}
\end{frame}

            \vspace{1em}
            $\Rightarrow$ \textbf{Idealized conditions}
        \end{column}
    \end{columns}
    
    \vspace{1em}
    \begin{block}{Foundation for Methodological Choices}

    \end{block}
